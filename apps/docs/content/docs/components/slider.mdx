## Installation

```bash
npx shadcn@latest add slider
```

## Usage

```tsx
import { Slider } from "@/components/ui/slider"

export function SliderDemo() {
  return <Slider defaultValue={[50]} max={100} step={1} className="w-full max-w-sm" />
}
```
